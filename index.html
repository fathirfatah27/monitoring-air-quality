#include #include #include "DHT.h" // =============================== // DATA WIFI // =============================== const char* ssid = "Hi"; const char* password = "anakayam"; // =============================== // DATA THINGSPEAK // =============================== unsigned long channelID = 3400896; const char* writeAPIKey = "2H0E0ZG35S7ZTGV8"; // =============================== // PIN SENSOR // =============================== #define DHTPIN D2 #define DHTTYPE DHT22 #define MQ135_PIN D5 #define MQ2_PIN D6 #define GP2Y_ANALOG A0 #define GP2Y_LED D7 #define BUZZER D1 DHT dht(DHTPIN, DHTTYPE); WiFiClient client; // =============================== // PENGATURAN SISTEM // =============================== float batasDebu = 150.0; // Waktu pemanasan awal 5 menit const unsigned long WAKTU_PEMANASAN = 300000UL; unsigned long waktuMulai; bool sensorSiap = false; // =============================== // KONEKSI WIFI // =============================== void koneksiWiFi() { WiFi.mode(WIFI_STA); WiFi.begin(ssid, password); Serial.println(); Serial.println("Menghubungkan ke WiFi..."); int percobaan = 0; while (WiFi.status() != WL_CONNECTED && percobaan < 40) { delay(500); Serial.print("."); percobaan++; } Serial.println(); if (WiFi.status() == WL_CONNECTED) { Serial.println("WiFi terhubung"); Serial.print("IP Address: "); Serial.println(WiFi.localIP()); } else { Serial.println("WiFi gagal terhubung"); } } // =============================== // BACA GP2Y SEKALI // =============================== int bacaAnalogGP2Y() { digitalWrite(GP2Y_LED, LOW); delayMicroseconds(280); int nilaiAnalog = analogRead(GP2Y_ANALOG); delayMicroseconds(40); digitalWrite(GP2Y_LED, HIGH); delayMicroseconds(9680); return nilaiAnalog; } // =============================== // BACA GP2Y RATA-RATA // =============================== float bacaGP2Y() { long totalAnalog = 0; for (int i = 0; i < 10; i++) { totalAnalog += bacaAnalogGP2Y(); delay(10); } float analogRata = totalAnalog / 10.0; // Sesuaikan dengan rentang tegangan A0 // pada board ESP8266 yang digunakan float tegangan = analogRata * (3.3 / 1023.0); // Estimasi konsentrasi debu float debu = (0.17 * tegangan - 0.1) * 1000.0; if (debu < 0) { debu = 0; } Serial.print("Analog GP2Y : "); Serial.println(analogRata); Serial.print("Tegangan GP2Y: "); Serial.print(tegangan); Serial.println(" V"); return debu; } // =============================== // SETUP // =============================== void setup() { Serial.begin(115200); delay(1000); // Inisialisasi sensor dht.begin(); pinMode(MQ135_PIN, INPUT); pinMode(MQ2_PIN, INPUT); pinMode(GP2Y_LED, OUTPUT); digitalWrite(GP2Y_LED, HIGH); // Buzzer mati saat sistem mulai pinMode(BUZZER, OUTPUT); digitalWrite(BUZZER, LOW); koneksiWiFi(); ThingSpeak.begin(client); // Mulai menghitung waktu pemanasan waktuMulai = millis(); Serial.println("Sistem mulai..."); Serial.println("Pemanasan sensor selama 5 menit."); Serial.println("Buzzer dinonaktifkan selama pemanasan."); } // =============================== // LOOP // =============================== void loop() { // Periksa koneksi WiFi if (WiFi.status() != WL_CONNECTED) { Serial.println("WiFi terputus, mencoba menghubungkan ulang..."); koneksiWiFi(); } // Periksa apakah waktu pemanasan selesai if (!sensorSiap && millis() - waktuMulai >= WAKTU_PEMANASAN) { sensorSiap = true; Serial.println("Pemanasan selesai."); Serial.println("Pemantauan bahaya diaktifkan."); } // Baca DHT22 float suhu = dht.readTemperature(); float kelembapan = dht.readHumidity(); if (isnan(suhu) || isnan(kelembapan)) { Serial.println("Gagal membaca sensor DHT!"); digitalWrite(BUZZER, LOW); delay(2000); return; } // Baca sensor gas digital int baca_udara = digitalRead(MQ135_PIN); int baca_asap = digitalRead(MQ2_PIN); // Mempertahankan logika dari kode awal: // LOW dianggap mendeteksi gas/bahaya int udara = (baca_udara == LOW) ? 1 : 0; int asap = (baca_asap == LOW) ? 1 : 0; // Baca sensor debu float debu = bacaGP2Y(); // =============================== // STATUS BAHAYA // =============================== int statusBahaya = 0; if (sensorSiap) { if (udara == 1 || asap == 1 || debu > batasDebu) { statusBahaya = 1; } } else { statusBahaya = 0; } // =============================== // KENDALI BUZZER // =============================== if (!sensorSiap) { // Selama 5 menit pertama, buzzer mati digitalWrite(BUZZER, LOW); } else if (statusBahaya == 1) { // Buzzer berbunyi putus-putus digitalWrite(BUZZER, HIGH); delay(500); digitalWrite(BUZZER, LOW); delay(500); } else { // Udara aman, buzzer mati digitalWrite(BUZZER, LOW); } // =============================== // TAMPILKAN DATA // =============================== Serial.println(); Serial.println("===== DATA SENSOR ====="); Serial.print("Suhu : "); Serial.print(suhu); Serial.println(" C"); Serial.print("Kelembapan : "); Serial.print(kelembapan); Serial.println(" %"); Serial.print("MQ135 : "); Serial.println(udara); Serial.print("MQ2 : "); Serial.println(asap); Serial.print("Debu GP2Y : "); Serial.print(debu); Serial.println(" ug/m3"); Serial.print("Sensor siap : "); Serial.println(sensorSiap ? "YA" : "BELUM"); Serial.print("Status : "); Serial.println(statusBahaya); if (!sensorSiap) { Serial.println("PEMANASAN SENSOR - ALARM NONAKTIF"); } else if (statusBahaya == 1) { Serial.println("BAHAYA - KUALITAS UDARA BURUK!"); } else { Serial.println("Udara aman"); } // =============================== // KIRIM DATA THINGSPEAK // =============================== ThingSpeak.setField(1, suhu); ThingSpeak.setField(2, kelembapan); ThingSpeak.setField(3, udara); ThingSpeak.setField(4, asap); ThingSpeak.setField(5, 0); ThingSpeak.setField(6, debu); ThingSpeak.setField(7, statusBahaya); Serial.println(); Serial.println("Mengirim data ke ThingSpeak..."); if (WiFi.status() == WL_CONNECTED) { int response = ThingSpeak.writeFields( channelID, writeAPIKey ); if (response == 200) { Serial.println("Data berhasil dikirim ke ThingSpeak"); } else { Serial.print("Gagal mengirim data. Kode error: "); Serial.println(response); } } else { Serial.println("Data tidak terkirim: WiFi belum terhubung"); } Serial.println("========================"); delay(20000); }
